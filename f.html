create view public.client_progress as select c.id as client_id, c.folder_paid as carpeta_done, c.initial_status = 'paid'::text as entrega_inicial_done, COALESCE(p.paid_count, 0::bigint) as cuotas_pagadas, COALESCE(p.total_count, c.installment_count::bigint) as cuotas_total, case when c.folder_paid then 1 else 0 end + case when c.initial_status = 'paid'::text then 1 else 0 end + COALESCE(p.paid_count, 0::bigint) as completados, 2 + COALESCE(p.total_count, c.installment_count::bigint) as total_compromisos, round( ( case when c.folder_paid then 1 else 0 end + case when c.initial_status = 'paid'::text then 1 else 0 end + COALESCE(p.paid_count, 0::bigint) )::numeric / NULLIF( 2 + COALESCE(p.total_count, c.installment_count::bigint), 0 )::numeric * 100::numeric ) as porcentaje, c.initial_status = 'paid'::text and COALESCE(p.total_count, 0::bigint) > 0 and COALESCE(p.paid_count, 0::bigint) = COALESCE(p.total_count, 0::bigint) as retiro_listo from clients c left join ( select payments.client_id, count(*) as total_count, count(*) filter ( where payments.status = 'paid'::text ) as paid_count from payments group by payments.client_id ) p on p.client_id = c.id;
